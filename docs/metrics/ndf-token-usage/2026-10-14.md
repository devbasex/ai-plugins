```bash
python3 scripts/token-usage-snapshot.py --released 10.17.75 --until 2026-10-14T23:31:00Z --min-version 10.17.73
python3 scripts/token-usage.py --min-version 10.17.73 --until 2026-10-14T23:31:00Z --by version   # 末尾の「集計の出力」
```
