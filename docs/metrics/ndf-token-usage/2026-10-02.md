```bash
python3 scripts/token-usage-snapshot.py --released 10.17.53 --until 2026-10-02T06:00:00Z --min-version 10.17.51
python3 scripts/token-usage.py --min-version 10.17.51 --until 2026-10-02T06:00:00Z --by version   # 末尾の「集計の出力」
```
