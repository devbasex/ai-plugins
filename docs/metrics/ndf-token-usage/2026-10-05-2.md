```bash
python3 scripts/token-usage-snapshot.py --released 10.17.62 --until 2026-10-05T19:30:00Z --min-version 10.17.60
python3 scripts/token-usage.py --min-version 10.17.60 --until 2026-10-05T19:30:00Z --by version   # 末尾の「集計の出力」
```
