```bash
python3 scripts/token-usage-snapshot.py --released 10.17.61 --until 2026-10-05T04:03:00Z --min-version 10.17.59
python3 scripts/token-usage.py --min-version 10.17.59 --until 2026-10-05T04:03:00Z --by version   # 末尾の「集計の出力」
```
