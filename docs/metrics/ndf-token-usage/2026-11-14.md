```bash
python3 scripts/token-usage-snapshot.py --released 10.27.63 --until 2026-11-14T01:09:00Z --min-version 10.27.61
python3 scripts/token-usage.py --min-version 10.27.61 --until 2026-11-14T01:09:00Z --by version   # 末尾の「集計の出力」
```
