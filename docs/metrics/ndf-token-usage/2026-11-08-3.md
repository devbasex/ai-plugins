```bash
python3 scripts/token-usage-snapshot.py --released 10.27.43 --until 2026-11-08T18:09:00Z --min-version 10.27.41
python3 scripts/token-usage.py --min-version 10.27.41 --until 2026-11-08T18:09:00Z --by version   # 末尾の「集計の出力」
```
