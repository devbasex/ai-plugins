```bash
python3 scripts/token-usage-snapshot.py --released 10.27.16 --until 2026-11-04T14:16:00Z --min-version 10.27.14
python3 scripts/token-usage.py --min-version 10.27.14 --until 2026-11-04T14:16:00Z --by version   # 末尾の「集計の出力」
```
