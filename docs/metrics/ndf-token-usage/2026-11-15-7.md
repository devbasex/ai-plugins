```bash
python3 scripts/token-usage-snapshot.py --released 10.27.72 --until 2026-11-15T18:30:00Z --min-version 10.27.70
python3 scripts/token-usage.py --min-version 10.27.70 --until 2026-11-15T18:30:00Z --by version   # 末尾の「集計の出力」
```
