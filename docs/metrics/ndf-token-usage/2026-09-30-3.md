```bash
python3 scripts/token-usage-snapshot.py --released 10.17.49 --until 2026-09-30T11:33:00Z --min-version 10.17.47
python3 scripts/token-usage.py --min-version 10.17.47 --until 2026-09-30T11:33:00Z --by version   # 末尾の「集計の出力」
```
