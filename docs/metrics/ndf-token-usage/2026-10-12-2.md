```bash
python3 scripts/token-usage-snapshot.py --released 10.17.72 --until 2026-10-12T16:05:00Z --min-version 10.17.70
python3 scripts/token-usage.py --min-version 10.17.70 --until 2026-10-12T16:05:00Z --by version   # 末尾の「集計の出力」
```
