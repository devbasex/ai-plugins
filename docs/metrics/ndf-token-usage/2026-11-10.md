```bash
python3 scripts/token-usage-snapshot.py --released 10.27.51 --until 2026-11-10T21:23:00Z --min-version 10.27.49
python3 scripts/token-usage.py --min-version 10.27.49 --until 2026-11-10T21:23:00Z --by version   # 末尾の「集計の出力」
```
