```bash
python3 scripts/token-usage-snapshot.py --released 10.17.65 --until 2026-10-06T15:02:00Z --min-version 10.17.62
python3 scripts/token-usage.py --min-version 10.17.62 --until 2026-10-06T15:02:00Z --by version   # 末尾の「集計の出力」
```
