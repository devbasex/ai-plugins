```bash
python3 scripts/token-usage-snapshot.py --released 10.17.57 --until 2026-10-03T05:10:00Z --min-version 10.17.55
python3 scripts/token-usage.py --min-version 10.17.55 --until 2026-10-03T05:10:00Z --by version   # 末尾の「集計の出力」
```
