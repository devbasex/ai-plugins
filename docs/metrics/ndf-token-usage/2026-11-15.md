```bash
python3 scripts/token-usage-snapshot.py --released 10.27.66 --until 2026-11-15T02:32:00Z --min-version 10.27.64
python3 scripts/token-usage.py --min-version 10.27.64 --until 2026-11-15T02:32:00Z --by version   # 末尾の「集計の出力」
```
