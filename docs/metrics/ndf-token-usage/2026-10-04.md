```bash
python3 scripts/token-usage-snapshot.py --released 10.17.59 --until 2026-10-04T03:54:00Z --min-version 10.17.57
python3 scripts/token-usage.py --min-version 10.17.57 --until 2026-10-04T03:54:00Z --by version   # 末尾の「集計の出力」
```
